    private static DatasetResult processDataset(String name, String path,
                                                Set<String> dropCols, boolean balanced) {
        DatasetResult result = new DatasetResult();
        result.name = name;
        try {
            Prepared p = prepareDataset(name, path, dropCols, balanced);
            result.prep = p;
            double[][] centroidU0 = buildStatePreparationUnitary(p.centroidState0);
            double[][] centroidU1 = buildStatePreparationUnitary(p.centroidState1);
            ScoreModel model1 = new Alg1Model(centroidU0, centroidU1);
            result.eval1 = evaluate("Algorithm 1 (UU-dagger with state-preparation unitaries)", model1, p);

            result.report2 = buildAlg2Report(p);
            final MultiQubitEnc.Circuit circuit0 = MultiQubitEnc.encode(p.centroidState0, QUBITS);
            final MultiQubitEnc.Circuit circuit1 = MultiQubitEnc.encode(p.centroidState1, QUBITS);
            ScoreModel modelCombined = new CombinedModel(circuit0, circuit1);
            result.evalCombined = evaluate("Algorithm 2 circuits + Algorithm 1 UU-dagger",
                    modelCombined, p);

            
            final Encoded enc = encodeAll(p);
            result.enc = enc;
            result.circuit0 = circuit0;
            result.circuit1 = circuit1;

            int dim = 2 * Parameter.paramCount(QUBITS);       
            double[] init = randomTheta(dim, SEED + 11);
            result.evalCombined3 = evaluateQfdnn("Algorithm 1 + 2 + 3 (PQC angles random, not trained)",
                    enc, init, circuit0, circuit1, p);

            System.out.println("  " + name + " : training QFDNN with COBYLA ...");
            result.training4 = trainModel(init, new ScoreFn() {
                public double[] scores(double[] theta) {
                    return diffScores(qfdnnProbs(enc.trainStates, theta, circuit0, circuit1));
                }
            }, p.trainY, SEED + 33);
            result.eval4 = evaluateQfdnn("Algorithm 4 : QFDNN (U + PQC + U-dagger + PQC, COBYLA trained)",
                    enc, result.training4.theta, circuit0, circuit1, p);

            // Train a second model with the actual ZGATRO objective on the SAME training split.
            // The objective combines clean margin loss with six-channel, three-scale ZNE loss.
            System.out.println("  " + name + " : training QFDNN with ZGATRO (ZNE + trust region) ...");
            // ZGATRO refines the COBYLA-trained parameters rather than restarting randomly.
            // This tests whether noise-guided refinement improves an already trained baseline.
            Novelty.Result zres = Novelty.optimize(result.training4.theta, new Novelty.Objective() {
                public double value(double[] theta) {
                    return zgatroObjective(theta, enc, p.trainY, circuit0, circuit1);
                }
            }, 15, 0.01, 0.03, 0.001);
            result.zgatroOptimization = zres;
            result.evalZGATRO = evaluateQfdnn("Algorithm 4 : QFDNN (ZGATRO trained)",
                    enc, zres.theta, circuit0, circuit1, p);
            result.zgatroNoiseAcc = noiseSweep(p, enc, zres.theta, result.evalZGATRO.threshold,
                    circuit0, circuit1);
            double[] initVar = randomTheta(Parameter.paramCount(QUBITS), SEED + 22);
            result.trainingVar = trainModel(initVar, new ScoreFn() {
                public double[] scores(double[] theta) {
                    return variationalScores(enc.trainStates, theta);
                }
            }, p.trainY, SEED + 44);
            double[] varTest = variationalScores(enc.testStates, result.trainingVar.theta);
            double[][] varProbs = new double[varTest.length][2];
            for (int i = 0; i < varTest.length; i++) {
                varProbs[i][0] = 1.0 - varTest[i];
                varProbs[i][1] = varTest[i];
            }
            result.evalVar = evaluateScores("Variational part only (no UU-dagger)",
                    variationalScores(enc.trainStates, result.trainingVar.theta), varTest, varProbs, p);

            result.classical = new ArrayList<Evaluation>();
            result.classical.add(fromPredictions("Logistic Regression",
                    logisticRegression(p.trainPCA, p.trainY, p.testPCA), p.testY));
            result.classical.add(fromPredictions("KNN (k=5)",
                    knn(p.trainPCA, p.trainY, p.testPCA, 5), p.testY));
            result.classical.add(fromPredictions("Random Forest",
                    randomForest(p.trainPCA, p.trainY, p.testPCA, 50, SEED), p.testY));

            System.out.println("  " + name + " : noise study ...");
            result.noiseAcc = noiseSweep(p, enc, result.training4.theta, result.eval4.threshold,
                    circuit0, circuit1);

            System.out.println("  " + name + " : done");
        } catch (Exception e) {
            result.error = e.getMessage();
            System.out.println("  " + name + " : ERROR - " + e.getMessage());
            e.printStackTrace();
        }
        return result;
    }

    private static Prepared prepareDataset(String name, String path,
                                           Set<String> dropCols, boolean balanced) throws IOException {
        Prepared p = new Prepared();
        p.name = name;

        DataSet data = loadCSV(path, dropCols);
        p.targetName = data.targetName;
        p.label0 = data.c0;
        p.label1 = data.c1;
        p.loadedTotal = data.x.length;
        p.loadedFeatures = data.x[0].length;
        p.loaded0 = count(data.y, 0);
        p.loaded1 = count(data.y, 1);

        data = samplePaperStyle(data, balanced, MAX_SAMPLES, SEED);
        p.used0 = count(data.y, 0);
        p.used1 = count(data.y, 1);

        Split sp = stratifiedSplit(data.x, data.y, TEST_FRACTION, SEED);
        p.trainBefore0 = count(sp.trY, 0);
        p.trainBefore1 = count(sp.trY, 1);
        p.test0 = count(sp.teY, 0);
        p.test1 = count(sp.teY, 1);

        Standardizer scaler = Standardizer.fit(sp.trX);
        scaler.transform(sp.trX);
        scaler.transform(sp.teX);

        int sizeBefore = sp.trX.length;
        sp = smote(sp, SMOTE_K, SEED);
        p.synthetic = sp.trX.length - sizeBefore;
        p.trainAfter0 = count(sp.trY, 0);
        p.trainAfter1 = count(sp.trY, 1);

        PCA pca = PCA.fit(sp.trX, PCA_COMPONENTS);
        double[][] trainPCA = pca.transform(sp.trX);
        double[][] testPCA = pca.transform(sp.teX);

        double[] sigma = columnStd(trainPCA);
        toUnitRange(trainPCA, sigma);
        toUnitRange(testPCA, sigma);

        p.pcaDims = trainPCA[0].length;
        p.trainPCA = trainPCA;
        p.testPCA = testPCA;
        p.trainY = sp.trY;
        p.testY = sp.teY;

        double[][] centroids = kmeansCentroids(trainPCA, sp.trY);
        p.centroidState0 = amplitudeEncode(centroids[0]);
        p.centroidState1 = amplitudeEncode(centroids[1]);
        p.centroidOverlap = dot(p.centroidState0, p.centroidState1);

        return p;
    }

    private static double[][] buildStatePreparationUnitary(double[] state) {
        double[] v = new double[STATE_SIZE];
        for (int i = 0; i < STATE_SIZE; i++) {
            v[i] = -state[i];
        }
        v[0] = v[0] + 1.0;
        double vNormSquared = dot(v, v);

        double[][] U = new double[STATE_SIZE][STATE_SIZE];
        for (int i = 0; i < STATE_SIZE; i++) {
            U[i][i] = 1.0;
        }
        if (vNormSquared > EPS) {
            double factor = 2.0 / vNormSquared;
            for (int i = 0; i < STATE_SIZE; i++) {
                for (int j = 0; j < STATE_SIZE; j++) {
                    U[i][j] = U[i][j] - factor * v[i] * v[j];
                }
            }
        }
        return U;
    }

    private static double alg1Amplitude(double[][] testUnitary, double[][] centroidUnitary) {
        double[][] A = multiplyMatrices(transpose(testUnitary), centroidUnitary);
        double[] zero = new double[STATE_SIZE];
        zero[0] = 1.0;
        double[] result = multiplyVector(A, zero);
        return result[0];
    }

    private interface ScoreModel {
        double amplitude(double[] testState, int classIndex);
    }

    private static class Alg1Model implements ScoreModel {
        private final double[][] centroidU0;
        private final double[][] centroidU1;

        Alg1Model(double[][] centroidU0, double[][] centroidU1) {
            this.centroidU0 = centroidU0;
            this.centroidU1 = centroidU1;
        }

        public double amplitude(double[] testState, int classIndex) {
            double[][] testUnitary = buildStatePreparationUnitary(testState);
            if (classIndex == 0) {
                return alg1Amplitude(testUnitary, centroidU0);
            }
            return alg1Amplitude(testUnitary, centroidU1);
        }
    }

    private static class CombinedModel implements ScoreModel {
        private final MultiQubitEnc.Circuit centroidCircuit0;
        private final MultiQubitEnc.Circuit centroidCircuit1;

        CombinedModel(MultiQubitEnc.Circuit c0, MultiQubitEnc.Circuit c1) {
            this.centroidCircuit0 = c0;
            this.centroidCircuit1 = c1;
        }

        public double amplitude(double[] testState, int classIndex) {
            MultiQubitEnc.Circuit testCircuit = MultiQubitEnc.encode(testState, QUBITS);
            if (classIndex == 0) {
                return MultiQubitEnc.overlapAmplitude(testCircuit, centroidCircuit0);
            }
            return MultiQubitEnc.overlapAmplitude(testCircuit, centroidCircuit1);
        }
    }

    private static Evaluation evaluate(String method, ScoreModel model, Prepared p) {

        Evaluation ev = new Evaluation();
        ev.method = method;

        int trainCount = p.trainPCA.length;
        double[] trainScore = new double[trainCount];

        for (int i = 0; i < trainCount; i++) {
            double[] state = amplitudeEncode(p.trainPCA[i]);
            double a0 = model.amplitude(state, 0);
            double a1 = model.amplitude(state, 1);
            trainScore[i] = a1 * a1 - a0 * a0;
        }

        ev.threshold = 0.0;
        if (TUNE_THRESHOLD) {
            ev.threshold = bestThreshold(trainScore, p.trainY);
        }

        int testCount = p.testPCA.length;
        ev.testCount = testCount;
        ev.confusion = new int[2][2];
        ev.probs = new double[testCount][2];
        ev.predicted = new int[testCount];
        ev.actual = new int[testCount];
        ev.maxDeviation = 0.0;

        for (int i = 0; i < testCount; i++) {
            double[] state = amplitudeEncode(p.testPCA[i]);
            double a0 = model.amplitude(state, 0);
            double a1 = model.amplitude(state, 1);

            double p0 = a0 * a0;
            double p1 = a1 * a1;
            double direct0 = sq(dot(p.centroidState0, state));
            double direct1 = sq(dot(p.centroidState1, state));
            ev.maxDeviation = Math.max(ev.maxDeviation, Math.abs(p0 - direct0));
            ev.maxDeviation = Math.max(ev.maxDeviation, Math.abs(p1 - direct1));

            int predicted = 0;
            if (p1 - p0 > ev.threshold) {
                predicted = 1;
            }

            ev.probs[i][0] = p0;
            ev.probs[i][1] = p1;
            ev.predicted[i] = predicted;
            ev.actual[i] = p.testY[i];
            ev.confusion[p.testY[i]][predicted]++;

            if (i == 0) {
                ev.firstState = state;
                ev.firstA0 = a0;
                ev.firstA1 = a1;
            }
        }

        computeMetrics(ev);
        return ev;
    }

    private static void computeMetrics(Evaluation ev) {
        int tn = ev.confusion[0][0];
        int fp = ev.confusion[0][1];
        int fn = ev.confusion[1][0];
        int tp = ev.confusion[1][1];
        ev.accuracy = (tp + tn) / (double) Math.max(1, ev.testCount);
        ev.precision = 0.0;
        if (tp + fp > 0) {
            ev.precision = tp / (double) (tp + fp);
        }
        ev.recall = 0.0;
        if (tp + fn > 0) {
            ev.recall = tp / (double) (tp + fn);
        }
        ev.f1 = 0.0;
        if (ev.precision + ev.recall > 0) {
            ev.f1 = 2.0 * ev.precision * ev.recall / (ev.precision + ev.recall);
        }
    }
    private static Alg2Report buildAlg2Report(Prepared p) {

        Alg2Report r = new Alg2Report();
        r.qubits = QUBITS;
        r.centroidCircuit0 = MultiQubitEnc.encode(p.centroidState0, QUBITS);
        r.centroidCircuit1 = MultiQubitEnc.encode(p.centroidState1, QUBITS);
        r.firstTarget = amplitudeEncode(p.testPCA[0]);
        r.firstCircuit = MultiQubitEnc.encode(r.firstTarget, QUBITS);
        r.firstEncoded = MultiQubitEnc.run(r.firstCircuit);

        double minFidelity = 1.0;
        double sumFidelity = 0.0;
        double maxError = 0.0;
        int checked = 0;

        List<MultiQubitEnc.Circuit> all = new ArrayList<MultiQubitEnc.Circuit>();
        all.add(r.centroidCircuit0);
        all.add(r.centroidCircuit1);
        for (int i = 0; i < p.testPCA.length; i++) {
            double[] state = amplitudeEncode(p.testPCA[i]);
            all.add(MultiQubitEnc.encode(state, QUBITS));
        }

        for (MultiQubitEnc.Circuit circuit : all) {
            double fidelity = MultiQubitEnc.fidelity(circuit);
            minFidelity = Math.min(minFidelity, fidelity);
            sumFidelity += fidelity;
            maxError = Math.max(maxError, MultiQubitEnc.maxAmplitudeError(circuit));
            checked++;
        }

        r.checked = checked;
        r.minFidelity = minFidelity;
        r.meanFidelity = sumFidelity / Math.max(1, checked);
        r.maxError = maxError;
        return r;
    }

    private static Encoded encodeAll(Prepared p) {
        Encoded e = new Encoded();
        e.trainStates = new double[p.trainPCA.length][];
        e.trainCircuits = new MultiQubitEnc.Circuit[p.trainPCA.length];
        for (int i = 0; i < p.trainPCA.length; i++) {
            // Keep the encoded training circuits so ZGATRO can evaluate its noisy objective.
            MultiQubitEnc.Circuit c = MultiQubitEnc.encode(amplitudeEncode(p.trainPCA[i]), QUBITS);
            e.trainCircuits[i] = c;
            e.trainStates[i] = MultiQubitEnc.run(c);
        }
        e.testStates = new double[p.testPCA.length][];
        e.testCircuits = new MultiQubitEnc.Circuit[p.testPCA.length];
        for (int i = 0; i < p.testPCA.length; i++) {
            MultiQubitEnc.Circuit c = MultiQubitEnc.encode(amplitudeEncode(p.testPCA[i]), QUBITS);
            e.testCircuits[i] = c;
            e.testStates[i] = MultiQubitEnc.run(c);
        }
        return e;
    }

    private static double[] randomTheta(int dim, long seed) {
        Random random = new Random(seed);
        double[] theta = new double[dim];
        for (int i = 0; i < dim; i++) {
            theta[i] = (2.0 * random.nextDouble() - 1.0) * INIT_RANGE;
        }
        return theta;
    }
    private static double[][] qfdnnProbs(double[][] states, double[] theta,
                                         MultiQubitEnc.Circuit c0, MultiQubitEnc.Circuit c1) {
        int half = Parameter.paramCount(QUBITS);
        Parameter.Circuit q1 = Parameter.constructCircuit(QUBITS, Arrays.copyOfRange(theta, 0, half));
        Parameter.Circuit q2 = Parameter.constructCircuit(QUBITS, Arrays.copyOfRange(theta, half, 2 * half));

        double[][] out = new double[states.length][2];
        for (int i = 0; i < states.length; i++) {
            out[i][0] = branchProbability(states[i], q1, c0, q2);
            out[i][1] = branchProbability(states[i], q1, c1, q2);
        }
        return out;
    }

    private static double branchProbability(double[] state, Parameter.Circuit q1,
                                            MultiQubitEnc.Circuit centroid, Parameter.Circuit q2) {
        double[] re = state.clone();
        double[] im = new double[state.length];
        Parameter.apply(q1, re, im);
        MultiQubitEnc.runInverse(re, centroid);      
        MultiQubitEnc.runInverse(im, centroid);
        Parameter.apply(q2, re, im);
        return re[0] * re[0] + im[0] * im[0];
    }

    private static double[] diffScores(double[][] probs) {
        double[] s = new double[probs.length];
        for (int i = 0; i < s.length; i++) {
            s[i] = probs[i][1] - probs[i][0];
        }
        return s;
    }
    private static double[] variationalScores(double[][] states, double[] theta) {
        Parameter.Circuit q = Parameter.constructCircuit(QUBITS, theta);
        int mask = 1 << (QUBITS - 1);
        double[] s = new double[states.length];
        for (int k = 0; k < states.length; k++) {
            double[] re = states[k].clone();
            double[] im = new double[re.length];
            Parameter.apply(q, re, im);
            double sum = 0.0;
            for (int i = 0; i < re.length; i++) {
                if ((i & mask) != 0) {
                    sum += re[i] * re[i] + im[i] * im[i];
                }
            }
            s[k] = sum;
        }
        return s;
    }

    private static double thresholdFor(double[] score, int[] labels) {
        if (TUNE_THRESHOLD) {
            return bestThreshold(score, labels);
        }
        return 0.0;
    }
    private static double lossOf(double[] score, int[] labels) {
        double thr = thresholdFor(score, labels);
        int ok = 0;
        for (int i = 0; i < score.length; i++) {
            int predicted = 0;
            if (score[i] > thr) {
                predicted = 1;
            }
            if (predicted == labels[i]) {
                ok++;
            }
        }
        return 1.0 - ok / (double) score.length;
    }

    private interface ScoreFn {
        double[] scores(double[] theta);
    }

    private static Training trainModel(final double[] theta0, final ScoreFn fn, final int[] labels, long seed) {

        Cobyla.Result r = Cobyla.minimize(new Cobyla.Objective() {
            public double value(double[] x) {
                return lossOf(fn.scores(x), labels);
            }
        }, theta0, COBYLA_RHO_BEGIN, COBYLA_RHO_END, MAX_EVALS, seed);

        Training t = new Training();
        t.theta0 = theta0.clone();
        t.theta = r.x;
        t.lossHistory = r.bestHistory;
        t.evaluations = r.evaluations;
        t.stop = r.stopReason;
        t.initialLoss = r.bestHistory.get(0);
        t.finalLoss = r.fx;
        return t;
    }

    private static Evaluation evaluateQfdnn(String method, Encoded enc, double[] theta,
                                            MultiQubitEnc.Circuit c0, MultiQubitEnc.Circuit c1, Prepared p) {
        double[][] trainP = qfdnnProbs(enc.trainStates, theta, c0, c1);
        double[][] testP = qfdnnProbs(enc.testStates, theta, c0, c1);
        return evaluateScores(method, diffScores(trainP), diffScores(testP), testP, p);
    }

    private static Evaluation evaluateScores(String method, double[] trainScore, double[] testScore,
                                             double[][] testProbs, Prepared p) {
        Evaluation ev = new Evaluation();
        ev.method = method;
        ev.threshold = thresholdFor(trainScore, p.trainY);
        int n = testScore.length;
        ev.testCount = n;
        ev.confusion = new int[2][2];
        ev.predicted = new int[n];
        ev.actual = new int[n];
        ev.probs = testProbs;
        for (int i = 0; i < n; i++) {
            int predicted = 0;
            if (testScore[i] > ev.threshold) {
                predicted = 1;
            }
            ev.predicted[i] = predicted;
            ev.actual[i] = p.testY[i];
            ev.confusion[p.testY[i]][predicted]++;
        }
        computeMetrics(ev);
        return ev;
    }

    private static Evaluation fromPredictions(String name, int[] predicted, int[] actual) {
        Evaluation ev = new Evaluation();
        ev.method = name;
        ev.testCount = actual.length;
        ev.confusion = new int[2][2];
        ev.predicted = predicted;
        ev.actual = actual;
        for (int i = 0; i < actual.length; i++) {
            ev.confusion[actual[i]][predicted[i]]++;
        }
        computeMetrics(ev);
        return ev;
    }

    /**
     * ZGATRO objective: clean differentiable margin loss + mean ZNE-mitigated
     * margin loss over all six channels. Uses only a fixed, deterministic subset
     * of training rows so finite-difference evaluations remain practical.
     */
    private static double zgatroObjective(double[] theta, Encoded enc, int[] labels,
                                           MultiQubitEnc.Circuit c0, MultiQubitEnc.Circuit c1) {
        final int[] sampleIndices = balancedObjectiveIndices(labels, 6);
        final int sampleCount = sampleIndices.length;
        final double[] scales = {1.0, 2.0, 3.0};
        final double baseNoise = 0.06;
        final double lambda = 0.65;
        final double beta = 0.05;
        int half = Parameter.paramCount(QUBITS);
        double[] th1 = Arrays.copyOfRange(theta, 0, half);
        double[] th2 = Arrays.copyOfRange(theta, half, theta.length);
        double cleanLoss = 0.0;
        for (int si = 0; si < sampleCount; si++) {
            int i = sampleIndices[si];
            double[][] probs = qfdnnProbs(new double[][]{enc.trainStates[i]}, theta, c0, c1);
            double score = probs[0][1] - probs[0][0];
            double target = labels[i] == 1 ? 1.0 : -1.0;
            double margin = Math.max(0.0, 0.5 - target * score);
            cleanLoss += margin * margin;
        }
        cleanLoss /= sampleCount;

        double noisyLoss = 0.0;
        double residualPenalty = 0.0;
        int channelCount = Parameter.NOISE_NAMES.length;
        for (int ch = 0; ch < channelCount; ch++) {
            for (int sample = 0; sample < sampleCount; sample++) {
                int i = sampleIndices[sample];
                double[] p0 = new double[3];
                double[] p1 = new double[3];
                for (int si = 0; si < scales.length; si++) {
                    // Common random numbers across parameter values stabilize finite differences.
                    long seed = 73471L + ch * 1009L + i * 97L + si * 31L;
                    Random rnd = new Random(seed);
                    double strength = Math.min(0.90, baseNoise * scales[si]);
                    p0[si] = Parameter.runNoisy(enc.trainCircuits[i], th1, c0, th2, ch, strength, rnd);
                    p1[si] = Parameter.runNoisy(enc.trainCircuits[i], th1, c1, th2, ch, strength, rnd);
                }
                double z0 = Novelty.extrapolateZeroNoise(scales, p0);
                double z1 = Novelty.extrapolateZeroNoise(scales, p1);
                double target = labels[i] == 1 ? 1.0 : -1.0;
                double margin = Math.max(0.0, 0.5 - target * (z1 - z0));
                noisyLoss += margin * margin;
                // Penalize scale-to-line disagreement: a larger residual means less reliable extrapolation.
                residualPenalty += Math.pow(p0[2] - (2.0 * p0[1] - p0[0]), 2)
                                 + Math.pow(p1[2] - (2.0 * p1[1] - p1[0]), 2);
            }
        }
        double denom = channelCount * sampleCount;
        noisyLoss /= denom;
        residualPenalty /= (2.0 * denom);
        return cleanLoss + lambda * noisyLoss + beta * residualPenalty;
    }

    private static int[] balancedObjectiveIndices(int[] labels, int perClass) {
        List<Integer> zero = new ArrayList<Integer>();
        List<Integer> one = new ArrayList<Integer>();
        for (int i = 0; i < labels.length; i++) {
            if (labels[i] == 0 && zero.size() < perClass) zero.add(i);
            else if (labels[i] == 1 && one.size() < perClass) one.add(i);
            if (zero.size() >= perClass && one.size() >= perClass) break;
        }
        List<Integer> selected = new ArrayList<Integer>();
        selected.addAll(zero); selected.addAll(one);
        if (selected.isEmpty()) throw new IllegalArgumentException("No training rows available for ZGATRO.");
        int[] out = new int[selected.size()];
        for (int i = 0; i < out.length; i++) out[i] = selected.get(i);
        return out;
    }

    private static double[][] noiseSweep(Prepared p, Encoded enc, double[] theta, double threshold,
                                         MultiQubitEnc.Circuit c0, MultiQubitEnc.Circuit c1) {

        int half = Parameter.paramCount(QUBITS);
        double[] th1 = Arrays.copyOfRange(theta, 0, half);
        double[] th2 = Arrays.copyOfRange(theta, half, 2 * half);
        int n = p.testPCA.length;
        Random rnd = new Random(SEED + 500);

        double[][] acc = new double[Parameter.NOISE_NAMES.length][NOISE_STEPS + 1];

        for (int t = 0; t < acc.length; t++) {
            for (int s = 0; s <= NOISE_STEPS; s++) {
                double prob = (double) s / NOISE_STEPS;
                int shots = (s == 0) ? 1 : NOISE_SHOTS;
                int ok = 0;
                for (int i = 0; i < n; i++) {
                    double s0 = 0.0;
                    double s1 = 0.0;
                    for (int k = 0; k < shots; k++) {
                        s0 += Parameter.runNoisy(enc.testCircuits[i], th1, c0, th2, t, prob, rnd);
                        s1 += Parameter.runNoisy(enc.testCircuits[i], th1, c1, th2, t, prob, rnd);
                    }
                    int predicted = 0;
                    if ((s1 - s0) / shots > threshold) {
                        predicted = 1;
                    }
                    if (predicted == p.testY[i]) {
                        ok++;
                    }
                }
                acc[t][s] = ok / (double) n;
            }
        }
        return acc;
    }
    private static int[] logisticRegression(double[][] X, int[] y, double[][] T) {
        int n = X.length;
        int d = X[0].length;
        double[] w = new double[d];
        double b = 0.0;
        double lr = 1.0;
        double lambda = 1.0e-3;

        for (int epoch = 0; epoch < 2000; epoch++) {
            double[] gw = new double[d];
            double gb = 0.0;
            for (int i = 0; i < n; i++) {
                double z = b;
                for (int j = 0; j < d; j++) {
                    z += w[j] * (X[i][j] - 0.5);
                }
                double err = 1.0 / (1.0 + Math.exp(-z)) - y[i];
                for (int j = 0; j < d; j++) {
                    gw[j] += err * (X[i][j] - 0.5);
                }
                gb += err;
            }
            for (int j = 0; j < d; j++) {
                w[j] -= lr * (gw[j] / n + lambda * w[j]);
            }
            b -= lr * gb / n;
        }

        int[] out = new int[T.length];
        for (int i = 0; i < T.length; i++) {
            double z = b;
            for (int j = 0; j < d; j++) {
                z += w[j] * (T[i][j] - 0.5);
            }
            if (z > 0.0) {
                out[i] = 1;
            }
        }
        return out;
    }

    private static int[] knn(double[][] X, int[] y, double[][] T, int k) {
        int[] out = new int[T.length];
        for (int i = 0; i < T.length; i++) {
            final double[] distance = new double[X.length];
            Integer[] order = new Integer[X.length];
            for (int j = 0; j < X.length; j++) {
                distance[j] = dist(T[i], X[j]);
                order[j] = j;
            }
            Arrays.sort(order, new Comparator<Integer>() {
                public int compare(Integer a, Integer b) {
                    return Double.compare(distance[a], distance[b]);
                }
            });
            int votes = 0;
            for (int m = 0; m < k; m++) {
                votes += y[order[m]];
            }
            if (votes * 2 > k) {
                out[i] = 1;
            }
        }
        return out;
    }

    private static class TreeNode {
        int feature = -1;
        double threshold;
        TreeNode left;
        TreeNode right;
        double prob;
    }

    private static int[] randomForest(double[][] X, int[] y, double[][] T, int trees, long seed) {
        Random random = new Random(seed + 900);
        List<TreeNode> forest = new ArrayList<TreeNode>();
        int n = X.length;
        for (int t = 0; t < trees; t++) {
            List<Integer> sample = new ArrayList<Integer>();
            for (int i = 0; i < n; i++) {
                sample.add(random.nextInt(n));
            }
            forest.add(buildTree(sample, X, y, 0, random));
        }
        int[] out = new int[T.length];
        for (int i = 0; i < T.length; i++) {
            double sum = 0.0;
            for (TreeNode root : forest) {
                TreeNode node = root;
                while (node.feature >= 0) {
                    if (T[i][node.feature] <= node.threshold) {
                        node = node.left;
                    } else {
                        node = node.right;
                    }
                }
                sum += node.prob;
            }
            if (sum / trees >= 0.5) {
                out[i] = 1;
            }
        }
        return out;
    }

    private static TreeNode buildTree(List<Integer> idx, final double[][] X, int[] y, int depth, Random random) {

        TreeNode node = new TreeNode();
        int pos = 0;
        for (int i : idx) {
            pos += y[i];
        }
        node.prob = pos / (double) idx.size();

        if (pos == 0 || pos == idx.size() || depth >= 12 || idx.size() < 4) {
            return node;
        }

        int d = X[0].length;
        List<Integer> features = new ArrayList<Integer>();
        for (int j = 0; j < d; j++) {
            features.add(j);
        }
        Collections.shuffle(features, random);
        int tryCount = Math.max(1, (int) Math.sqrt(d));

        double bestGini = Double.MAX_VALUE;
        int bestFeature = -1;
        double bestThr = 0.0;

        for (int fi = 0; fi < tryCount; fi++) {
            final int f = features.get(fi);
            List<Integer> sorted = new ArrayList<Integer>(idx);
            Collections.sort(sorted, new Comparator<Integer>() {
                public int compare(Integer a, Integer b) {
                    return Double.compare(X[a][f], X[b][f]);
                }
            });
            int total = sorted.size();
            int leftPos = 0;
            for (int k = 0; k < total - 1; k++) {
                leftPos += y[sorted.get(k)];
                double a = X[sorted.get(k)][f];
                double b = X[sorted.get(k + 1)][f];
                if (a >= b) {
                    continue;
                }
                int leftN = k + 1;
                int rightN = total - leftN;
                int rightPos = pos - leftPos;
                double gl = 1.0 - sq(leftPos / (double) leftN) - sq(1.0 - leftPos / (double) leftN);
                double gr = 1.0 - sq(rightPos / (double) rightN) - sq(1.0 - rightPos / (double) rightN);
                double gini = (leftN * gl + rightN * gr) / total;
                if (gini < bestGini) {
                    bestGini = gini;
                    bestFeature = f;
                    bestThr = 0.5 * (a + b);
                }
            }
        }

        if (bestFeature < 0) {
            return node;
        }

        List<Integer> left = new ArrayList<Integer>();
        List<Integer> right = new ArrayList<Integer>();
        for (int i : idx) {
            if (X[i][bestFeature] <= bestThr) {
                left.add(i);
            } else {
                right.add(i);
            }
        }
        if (left.isEmpty() || right.isEmpty()) {
            return node;
        }

        node.feature = bestFeature;
        node.threshold = bestThr;
        node.left = buildTree(left, X, y, depth + 1, random);
        node.right = buildTree(right, X, y, depth + 1, random);
        return node;
    }

    private static double meanNoiseError(double[][] noise) {
        if (noise == null || noise.length == 0) return Double.NaN;
        double sum = 0; int n = 0;
        for (double[] channel : noise) for (int i = 1; i < channel.length; i++) { sum += 1.0 - channel[i]; n++; }
        return n == 0 ? Double.NaN : sum / n;
    }

    private static double[][] combineNoise(double[][] a, double[][] b) {
        double[][] out = new double[a.length + b.length][];
        for (int i = 0; i < a.length; i++) out[i] = a[i];
        for (int i = 0; i < b.length; i++) out[a.length + i] = b[i];
        return out;
    }

private static void calculateLiveStatisticalRows(StringBuilder sb, DatasetResult dr) {
    Evaluation qfdnn = dr.eval4;
    int n = qfdnn.testCount;

    for (Evaluation classical : dr.classical) {
        double p1 = classical.accuracy;
        double p2 = qfdnn.accuracy;
        double v1 = (p1 * (1.0 - p1)) / n;
        double v2 = (p2 * (1.0 - p2)) / n;
        double standardError = Math.sqrt(v1 + v2);

        double tStat = 0.0;
        double pVal = 1.0;
        String isSig = "No";

        if (standardError > 1e-9) {
            tStat = (p2 - p1) / standardError;
            double absT = Math.abs(tStat);
            pVal = Math.exp(-0.717 * absT - 0.416 * absT * absT);
            if (pVal < 0.05) isSig = "Yes";
        } else if (Math.abs(p1 - p2) > 1e-4) {
            tStat = p2 > p1 ? Double.POSITIVE_INFINITY : Double.NEGATIVE_INFINITY;
            pVal = 0.0000;
            isSig = "Yes";
        }

        String tStr = Double.isInfinite(tStat) ? (tStat > 0 ? "+&infin;" : "&minus;&infin;") : f(tStat, 4);
        String pStr = pVal < 0.0001 ? "0.0000" : f(pVal, 4);
        String shortLabel = classical.method;
        if (shortLabel.toUpperCase().contains("LOGISTIC")) shortLabel = "LR";
        else if (shortLabel.toUpperCase().contains("KNN")) shortLabel = "KNN";
        else if (shortLabel.toUpperCase().contains("FOREST")) shortLabel = "RF";

        sb.append("<tr><td>").append(shortLabel).append("</td><td>").append(tStr)
          .append("</td><td>").append(pStr).append("</td><td>").append(isSig).append("</td></tr>");
    }
}

    private static String trainingChart(Training t, String title) {
        int n = t.lossHistory.size();
        double[] xs = new double[n];
        double[] acc = new double[n];
        double[] loss = new double[n];
        for (int i = 0; i < n; i++) {
            xs[i] = i + 1;
            loss[i] = t.lossHistory.get(i);
            acc[i] = 1.0 - loss[i];
        }
        return lineChartSvg(title, "Cost evaluation", "Value", xs,
                new String[] {"Accuracy", "Loss"}, new double[][] {acc, loss}, 0.0, 1.0);
    }

    private static String heatmap(String title, int[][] cm) {
        int total = 0;
        int max = 1;
        for (int[] row : cm) {
            for (int v : row) {
                total += v;
                max = Math.max(max, v);
            }
        }
        total = Math.max(1, total);

        StringBuilder s = new StringBuilder();
        s.append("<div><h4>").append(h(title)).append("</h4>");
        s.append("<table class='cm'><tr><th></th><th>Predicted 0</th><th>Predicted 1</th></tr>");
        for (int r = 0; r < 2; r++) {
            s.append("<tr><th>Actual ").append(r).append("</th>");
            for (int c = 0; c < 2; c++) {
                double t = cm[r][c] / (double) max;
                int red = (int) Math.round(255 + (8 - 255) * t);
                int green = (int) Math.round(255 + (48 - 255) * t);
                int blue = (int) Math.round(255 + (107 - 255) * t);
                String text = (t > 0.5) ? "#ffffff" : "#1c2333";

    private static void startServer(Map<String, String> pages) {

        HttpServer server = null;
        int port = START_PORT;

        while (server == null && port <= START_PORT + 10) {
            try {
                server = HttpServer.create(new InetSocketAddress(port), 0);
            } catch (IOException e) {
                port++;
            }
        }

        if (server == null) {
            System.err.println("Could not start the web server (ports " + START_PORT + "-"
                    + (START_PORT + 10) + " are busy).");
            return;
        }

        final String page = buildPage();
        server.createContext("/", exchange -> sendText(exchange, page));
        for (final Map.Entry<String, String> entry : pages.entrySet()) {
            server.createContext("/result/" + entry.getKey(), exchange -> sendText(exchange, entry.getValue()));
        }

        server.setExecutor(null);
        server.start();

        String url = "http://localhost:" + port;

        System.out.println();
        System.out.println("Web page is ready  ->  " + url);
        System.out.println("Open it in your browser and use the buttons.");
        System.out.println("(Press Ctrl+C to quit.)");

        try {
            if (Desktop.isDesktopSupported()) {
                Desktop.getDesktop().browse(new URI(url));
            }
        } catch (Exception e) {
        } catch (java.awt.AWTError e) {

        }
    }

    private static void sendText(com.sun.net.httpserver.HttpExchange exchange, String body) throws IOException {
        byte[] bytes = body.getBytes(StandardCharsets.UTF_8);
        exchange.getResponseHeaders().set("Content-Type", "text/html; charset=utf-8");
        exchange.sendResponseHeaders(200, bytes.length);
        try (OutputStream os = exchange.getResponseBody()) {
            os.write(bytes);
        }
    }

    private static double bestThreshold(double[] score, int[] labels) {

        int n = score.length;
        final double[] s = score;

        Integer[] order = new Integer[n];
        for (int i = 0; i < n; i++) {
            order[i] = i;
        }

        Arrays.sort(order, (a, b) -> Double.compare(s[a], s[b]));

        int correct = count(labels, 1);
        int bestCorrect = correct;
        double bestThreshold = score[order[0]] - 1.0e-9;

        for (int k = 0; k < n; k++) {
            if (labels[order[k]] == 0) {
                correct++;
            } else {
                correct--;
            }

            boolean lastOne = (k == n - 1);
            boolean valueChanges = false;
            if (!lastOne) {
                valueChanges = score[order[k]] < score[order[k + 1]];
            }

            if (lastOne || valueChanges) {
                if (correct > bestCorrect) {
                    bestCorrect = correct;
                    if (lastOne) {
                        bestThreshold = score[order[k]] + 1.0e-9;
                    } else {
                        bestThreshold = 0.5 * (score[order[k]] + score[order[k + 1]]);
                    }
                }
            }
        }

        return bestThreshold;
    }

    private static double[][] kmeansCentroids(double[][] X, int[] labels) {

        int n = X.length;
        int d = X[0].length;
        double[][] centers = new double[2][d];
        int[] classCount = new int[2];

        for (int i = 0; i < n; i++) {
            int c = labels[i];
            classCount[c]++;
            for (int j = 0; j < d; j++) {
                centers[c][j] += X[i][j];
            }
        }

        for (int k = 0; k < 2; k++) {
            for (int j = 0; j < d; j++) {
                centers[k][j] = centers[k][j] / Math.max(1, classCount[k]);
            }
        }

        for (int iteration = 0; iteration < KMEANS_MAX_ITER; iteration++) {

            double[][] newCenters = new double[2][d];
            int[] members = new int[2];

            for (int i = 0; i < n; i++) {
                int cluster = 1;
                if (dist(X[i], centers[0]) <= dist(X[i], centers[1])) {
                    cluster = 0;
                }
                members[cluster]++;
                for (int j = 0; j < d; j++) {
                    newCenters[cluster][j] += X[i][j];
                }
            }

            double movement = 0.0;

            for (int k = 0; k < 2; k++) {
                if (members[k] == 0) {
                    newCenters[k] = centers[k].clone();
                } else {
                    for (int j = 0; j < d; j++) {
                        newCenters[k][j] = newCenters[k][j] / members[k];
                    }
                }
                movement += dist(centers[k], newCenters[k]);
            }

            centers = newCenters;

            if (movement < 1.0e-10) {
                break;
            }
        }

        return centers;
    }

    private static double[] amplitudeEncode(double[] features) {

        double[] state = new double[STATE_SIZE];

        for (int i = 0; i < PCA_COMPONENTS; i++) {
            state[i] = features[i];
        }
        double length = norm(state);

        if (length < EPS) {
            state[0] = 1.0;
            return state;
        }

        for (int i = 0; i < STATE_SIZE; i++) {
            state[i] = state[i] / length;
        }

        return state;
    }
    private static DataSet loadCSV(String path, Set<String> dropCols) throws IOException {

        File file = new File(path);
        if (!file.exists()) {
            throw new IOException("CSV file not found: " + file.getAbsolutePath());
        }

        List<String[]> rows = new ArrayList<String[]>();
        String[] header;

        try (BufferedReader reader = new BufferedReader(new FileReader(file))) {

            String headerLine = reader.readLine();
            if (headerLine == null) {
                throw new IOException("CSV is empty");
            }

            header = parseCSV(headerLine);
            for (int i = 0; i < header.length; i++) {
                header[i] = clean(header[i]);
            }

            String line;
            while ((line = reader.readLine()) != null) {
                if (line.trim().isEmpty()) {
                    continue;
                }
                String[] fields = parseCSV(line);
                if (fields.length == header.length) {
                    rows.add(fields);
                }
            }
        }

        if (rows.isEmpty()) {
            throw new IOException("No data rows");
        }

        int targetCol = header.length - 1;
        String[] preferred = {"Class", "Loan_Status", "Target", "Label"};
        boolean found = false;

        for (int p = 0; p < preferred.length && !found; p++) {
            for (int i = 0; i < header.length; i++) {
                if (header[i].equalsIgnoreCase(preferred[p])) {
                    targetCol = i;
                    found = true;
                    break;
                }
            }
        }

        List<Integer> featureCols = new ArrayList<Integer>();

        for (int c = 0; c < header.length; c++) {
            String lower = header[c].toLowerCase(Locale.ROOT);
            if (c == targetCol) {
                continue;
            }
            if (isId(lower)) {
                continue;
            }
            if (dropCols.contains(lower)) {
                continue;
            }
            featureCols.add(c);
        }

        List<String> targetValues = new ArrayList<String>();

        for (String[] row : rows) {
            String value = clean(row[targetCol]);
            if (!missing(value) && !containsIgnoreCase(targetValues, value)) {
                targetValues.add(value);
            }
        }

        if (targetValues.size() != 2) {
            throw new IOException("Target must have 2 classes, found "
                    + targetValues.size() + " " + targetValues);
        }

        String class0;
        String class1;

        if (containsIgnoreCase(targetValues, "N") && containsIgnoreCase(targetValues, "Y")) {
            class0 = findIgnoreCase(targetValues, "N");
            class1 = findIgnoreCase(targetValues, "Y");
        } else if (containsIgnoreCase(targetValues, "0") && containsIgnoreCase(targetValues, "1")) {
            class0 = findIgnoreCase(targetValues, "0");
            class1 = findIgnoreCase(targetValues, "1");
        } else {
            Collections.sort(targetValues);
            class0 = targetValues.get(0);
            class1 = targetValues.get(1);
        }

        int featureCount = featureCols.size();
        boolean[] isNumeric = new boolean[featureCount];
        double[] fillValue = new double[featureCount];

        @SuppressWarnings("unchecked")
        Map<String, Integer>[] categoryCodes = new Map[featureCount];

        for (int j = 0; j < featureCount; j++) {

            int col = featureCols.get(j);
            int checked = 0;
            int numeric = 0;
            List<Double> numbers = new ArrayList<Double>();

            for (String[] row : rows) {
                String value = clean(row[col]);
                if (missing(value)) {
                    continue;
                }
                checked++;
                Double number = parseNumber(value);
                if (number != null) {
                    numeric++;
                    numbers.add(number);
                }
            }

            isNumeric[j] = (checked > 0) && (numeric / (double) checked >= 0.80);

            if (isNumeric[j]) {
                fillValue[j] = median(numbers);
            } else {
                categoryCodes[j] = new LinkedHashMap<String, Integer>();
                Map<Integer, Integer> frequency = new HashMap<Integer, Integer>();

                for (String[] row : rows) {
                    String value = clean(row[col]);
                    if (missing(value)) {
                        continue;
                    }
                    String key = value.toLowerCase(Locale.ROOT);
                    if (!categoryCodes[j].containsKey(key)) {
                        categoryCodes[j].put(key, categoryCodes[j].size());
                    }
                    int code = categoryCodes[j].get(key);
                    Integer old = frequency.get(code);
                    if (old == null) {
                        frequency.put(code, 1);
                    } else {
                        frequency.put(code, old + 1);
                    }
                }

                int modeCode = 0;
                int modeCount = -1;
                for (Map.Entry<Integer, Integer> entry : frequency.entrySet()) {
                    if (entry.getValue() > modeCount) {
                        modeCount = entry.getValue();
                        modeCode = entry.getKey();
                    }
                }
                fillValue[j] = modeCode;
            }
        }

        List<double[]> xList = new ArrayList<double[]>();
        List<Integer> yList = new ArrayList<Integer>();

        for (String[] row : rows) {

            String target = clean(row[targetCol]);
            int label;

            if (target.equalsIgnoreCase(class0)) {
                label = 0;
            } else if (target.equalsIgnoreCase(class1)) {
                label = 1;
            } else {
                continue;
            }

            double[] vector = new double[featureCount];

            for (int j = 0; j < featureCount; j++) {

                String text = clean(row[featureCols.get(j)]);

                if (missing(text)) {
                    vector[j] = fillValue[j];
                } else if (isNumeric[j]) {
                    Double number = parseNumber(text);
                    if (number == null) {
                        vector[j] = fillValue[j];
                    } else {
                        vector[j] = number;
                    }
                } else {
                    Integer code = categoryCodes[j].get(text.toLowerCase(Locale.ROOT));
                    if (code == null) {
                        vector[j] = fillValue[j];
                    } else {
                        vector[j] = code;
                    }
                }
            }

            xList.add(vector);
            yList.add(label);
        }

        double[][] X = xList.toArray(new double[0][]);
        int[] Y = new int[yList.size()];
        for (int i = 0; i < Y.length; i++) {
            Y[i] = yList.get(i);
        }

        X = removeConstantColumns(X);

        if (X[0].length < PCA_COMPONENTS) {
            throw new IOException("Only " + X[0].length + " usable features, need " + PCA_COMPONENTS);
        }

        return new DataSet(X, Y, header[targetCol], class0, class1);
    }

    private static DataSet samplePaperStyle(DataSet data, boolean balanced,
                                            int maxSamples, long seed) {

        if (maxSamples <= 0) {
            return data;
        }

        List<Integer> class0 = new ArrayList<Integer>();
        List<Integer> class1 = new ArrayList<Integer>();

        for (int i = 0; i < data.y.length; i++) {
            if (data.y[i] == 0) {
                class0.add(i);
            } else {
                class1.add(i);
            }
        }

        Random random = new Random(seed + 7);
        Collections.shuffle(class0, random);
        Collections.shuffle(class1, random);

        int take0;
        int take1;

        if (balanced) {
            int each = Math.min(maxSamples / 2, Math.min(class0.size(), class1.size()));
            take0 = each;
            take1 = each;
        } else {
            if (data.y.length <= maxSamples) {
                return data;
            }
            take0 = (int) Math.round(maxSamples * class0.size() / (double) data.y.length);
            take1 = maxSamples - take0;
        }

        List<Integer> chosen = new ArrayList<Integer>();
        chosen.addAll(class0.subList(0, Math.min(take0, class0.size())));
        chosen.addAll(class1.subList(0, Math.min(take1, class1.size())));

        double[][] X = new double[chosen.size()][];
        int[] Y = new int[chosen.size()];

        for (int i = 0; i < chosen.size(); i++) {
            X[i] = data.x[chosen.get(i)];
            Y[i] = data.y[chosen.get(i)];
        }

        return new DataSet(X, Y, data.targetName, data.c0, data.c1);
    }

    private static Split stratifiedSplit(double[][] X, int[] y, double testFraction, long seed) {

        List<Integer> class0 = new ArrayList<Integer>();
        List<Integer> class1 = new ArrayList<Integer>();

        for (int i = 0; i < y.length; i++) {
            if (y[i] == 0) {
                class0.add(i);
            } else {
                class1.add(i);
            }
        }

        Random random = new Random(seed);
        Collections.shuffle(class0, random);
        Collections.shuffle(class1, random);

        List<Integer> trainIdx = new ArrayList<Integer>();
        List<Integer> testIdx = new ArrayList<Integer>();

        splitOneClass(class0, testFraction, trainIdx, testIdx);
        splitOneClass(class1, testFraction, trainIdx, testIdx);

        Collections.shuffle(trainIdx, random);
        Collections.shuffle(testIdx, random);

        double[][] trX = new double[trainIdx.size()][];
        int[] trY = new int[trainIdx.size()];
        double[][] teX = new double[testIdx.size()][];
        int[] teY = new int[testIdx.size()];

        for (int i = 0; i < trainIdx.size(); i++) {
            trX[i] = X[trainIdx.get(i)].clone();
            trY[i] = y[trainIdx.get(i)];
        }
        for (int i = 0; i < testIdx.size(); i++) {
            teX[i] = X[testIdx.get(i)].clone();
            teY[i] = y[testIdx.get(i)];
        }

        return new Split(trX, trY, teX, teY);
    }

    private static void splitOneClass(List<Integer> indices, double testFraction,
                                      List<Integer> train, List<Integer> test) {

        int testCount = (int) Math.round(indices.size() * testFraction);

        if (indices.size() > 1) {
            testCount = Math.max(1, Math.min(indices.size() - 1, testCount));
        } else {
            testCount = 0;
        }

        for (int i = 0; i < indices.size(); i++) {
            if (i < testCount) {
                test.add(indices.get(i));
            } else {
                train.add(indices.get(i));
            }
        }
    }

    private static class Standardizer {

        final double[] mean;
        final double[] std;

        Standardizer(double[] mean, double[] std) {
            this.mean = mean;
            this.std = std;
        }

        static Standardizer fit(double[][] X) {

            int n = X.length;
            int d = X[0].length;
            double[] mean = new double[d];
            double[] std = new double[d];

            for (double[] row : X) {
                for (int j = 0; j < d; j++) {
                    mean[j] += row[j];
                }
            }
            for (int j = 0; j < d; j++) {
                mean[j] = mean[j] / n;
            }
            for (double[] row : X) {
                for (int j = 0; j < d; j++) {
                    std[j] += sq(row[j] - mean[j]);
                }
            }
            for (int j = 0; j < d; j++) {
                std[j] = Math.sqrt(std[j] / n);
                if (std[j] < EPS) {
                    std[j] = 1.0;
                }
            }
            return new Standardizer(mean, std);
        }

        void transform(double[][] X) {
            for (double[] row : X) {
                for (int j = 0; j < row.length; j++) {
                    row[j] = (row[j] - mean[j]) / std[j];
                }
            }
        }
    }

    private static Split smote(Split sp, int k, long seed) {

        List<double[]> class0 = new ArrayList<double[]>();
        List<double[]> class1 = new ArrayList<double[]>();

        for (int i = 0; i < sp.trX.length; i++) {
            if (sp.trY[i] == 0) {
                class0.add(sp.trX[i]);
            } else {
                class1.add(sp.trX[i]);
            }
        }

        if (class0.isEmpty() || class1.isEmpty() || class0.size() == class1.size()) {
            return sp;
        }

        List<double[]> minority;
        int minorityLabel;

        if (class0.size() < class1.size()) {
            minority = class0;
            minorityLabel = 0;
        } else {
            minority = class1;
            minorityLabel = 1;
        }

        int needed = Math.abs(class0.size() - class1.size());

        if (minority.size() < 2) {
            return sp;
        }

        int neighbours = Math.min(k, minority.size() - 1);
        int[][] nearest = new int[minority.size()][];
        for (int i = 0; i < minority.size(); i++) {
            nearest[i] = nearestNeighbours(minority, i, neighbours);
        }

        Random random = new Random(seed + 100);

        int oldSize = sp.trX.length;
        double[][] newX = Arrays.copyOf(sp.trX, oldSize + needed);
        int[] newY = Arrays.copyOf(sp.trY, oldSize + needed);

        for (int s = 0; s < needed; s++) {

            int baseIndex = random.nextInt(minority.size());
            double[] base = minority.get(baseIndex);

            int neighbourIndex = nearest[baseIndex][random.nextInt(neighbours)];
            double[] neighbour = minority.get(neighbourIndex);

            double lambda = random.nextDouble();

            double[] sample = new double[base.length];
            for (int j = 0; j < sample.length; j++) {
                sample[j] = base[j] + lambda * (neighbour[j] - base[j]);
            }

            newX[oldSize + s] = sample;
            newY[oldSize + s] = minorityLabel;
        }

        return new Split(newX, newY, sp.teX, sp.teY);
    }

    private static int[] nearestNeighbours(List<double[]> points, int index, int k) {

        int n = points.size();
        final double[] distance = new double[n];
        Integer[] order = new Integer[n];

        for (int i = 0; i < n; i++) {
            if (i == index) {
                distance[i] = Double.POSITIVE_INFINITY;
            } else {
                distance[i] = dist(points.get(index), points.get(i));
            }
            order[i] = i;
        }

        Arrays.sort(order, (a, b) -> Double.compare(distance[a], distance[b]));

        int[] result = new int[k];
        for (int i = 0; i < k; i++) {
            result[i] = order[i];
        }
        return result;
    }

    private static class PCA {

        final double[] mean;
        final double[][] components;
        final int count;

        PCA(double[] mean, double[][] components, int count) {
            this.mean = mean;
            this.components = components;
            this.count = count;
        }

        static PCA fit(double[][] X, int requested) {

            int n = X.length;
            int d = X[0].length;
            double[] mean = new double[d];
            for (double[] row : X) {
                for (int j = 0; j < d; j++) {
                    mean[j] += row[j];
                }
            }
            for (int j = 0; j < d; j++) {
                mean[j] = mean[j] / n;
            }

            double[][] cov = new double[d][d];
            for (double[] row : X) {
                for (int i = 0; i < d; i++) {
                    double di = row[i] - mean[i];
                    for (int j = i; j < d; j++) {
                        cov[i][j] += di * (row[j] - mean[j]);
                    }
                }
            }
            for (int i = 0; i < d; i++) {
                for (int j = i; j < d; j++) {
                    cov[i][j] = cov[i][j] / Math.max(1, n - 1);
                    cov[j][i] = cov[i][j];
                }
            }

            double[][] vectors = new double[d][d];
            final double[] values = jacobi(cov, vectors);

            Integer[] order = new Integer[d];
            for (int i = 0; i < d; i++) {
                order[i] = i;
            }
            Arrays.sort(order, (a, b) -> Double.compare(values[b], values[a]));

            int keep = Math.min(requested, d);
            double[][] comp = new double[keep][d];
            for (int c = 0; c < keep; c++) {
                for (int j = 0; j < d; j++) {
                    comp[c][j] = vectors[j][order[c]];
                }
            }
            return new PCA(mean, comp, keep);
        }

        double[][] transform(double[][] X) {
            double[][] result = new double[X.length][count];
            for (int i = 0; i < X.length; i++) {
                for (int c = 0; c < count; c++) {
                    double sum = 0.0;
                    for (int j = 0; j < mean.length; j++) {
                        sum += (X[i][j] - mean[j]) * components[c][j];
                    }
                    result[i][c] = sum;
                }
            }
            return result;
        }
    }

    private static double[] jacobi(double[][] input, double[][] V) {

        int n = input.length;
        double[][] A = new double[n][n];

        for (int i = 0; i < n; i++) {
            A[i] = input[i].clone();
            Arrays.fill(V[i], 0.0);
            V[i][i] = 1.0;
        }

        int maxIterations = 100 * n * n;

        for (int iteration = 0; iteration < maxIterations; iteration++) {
            int p = 0;
            int q = 1;
            double max = Math.abs(A[0][1]);

            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    if (Math.abs(A[i][j]) > max) {
                        max = Math.abs(A[i][j]);
                        p = i;
                        q = j;
                    }
                }
            }

            if (max < 1.0e-12) {
                break;
            }

            double theta = 0.5 * Math.atan2(2.0 * A[p][q], A[q][q] - A[p][p]);
            double c = Math.cos(theta);
            double s = Math.sin(theta);

            double app = A[p][p];
            double aqq = A[q][q];
            double apq = A[p][q];

            A[p][p] = c * c * app - 2.0 * s * c * apq + s * s * aqq;
            A[q][q] = s * s * app + 2.0 * s * c * apq + c * c * aqq;
            A[p][q] = 0.0;
            A[q][p] = 0.0;

            for (int r = 0; r < n; r++) {
                if (r == p || r == q) {
                    continue;
                }
                double arp = A[r][p];
                double arq = A[r][q];

                A[r][p] = c * arp - s * arq;
                A[p][r] = A[r][p];

                A[r][q] = s * arp + c * arq;
                A[q][r] = A[r][q];
            }

            for (int r = 0; r < n; r++) {
                double vp = V[r][p];
                double vq = V[r][q];
                V[r][p] = c * vp - s * vq;
                V[r][q] = s * vp + c * vq;
            }
        }

        double[] values = new double[n];
        for (int i = 0; i < n; i++) {
            values[i] = A[i][i];
        }
        return values;
    }

    private static double[] columnStd(double[][] X) {

        int n = X.length;
        int d = X[0].length;
        double[] mean = new double[d];
        double[] std = new double[d];

        for (double[] row : X) {
            for (int j = 0; j < d; j++) {
                mean[j] += row[j];
            }
        }
        for (int j = 0; j < d; j++) {
            mean[j] = mean[j] / n;
        }
        for (double[] row : X) {
            for (int j = 0; j < d; j++) {
                std[j] += sq(row[j] - mean[j]);
            }
        }
        for (int j = 0; j < d; j++) {
            std[j] = Math.sqrt(std[j] / n);
            if (std[j] < EPS) {
                std[j] = 1.0;
            }
        }
        return std;
    }

    private static void toUnitRange(double[][] X, double[] sigma) {
        for (double[] row : X) {
            for (int j = 0; j < row.length; j++) {
                double z = row[j] / sigma[j];
                z = Math.max(-PCA_CLIP, Math.min(PCA_CLIP, z));
                row[j] = (z + PCA_CLIP) / (2.0 * PCA_CLIP);
            }
        }
    }
    private static double[][] transpose(double[][] m) {
        int rows = m.length;
        int cols = m[0].length;
        double[][] result = new double[cols][rows];
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                result[j][i] = m[i][j];
            }
        }
        return result;
    }

    private static double[][] multiplyMatrices(double[][] a, double[][] b) {
        int rowsA = a.length;
        int colsA = a[0].length;
        int colsB = b[0].length;
        double[][] result = new double[rowsA][colsB];
        for (int i = 0; i < rowsA; i++) {
            for (int j = 0; j < colsB; j++) {
                double sum = 0.0;
                for (int k = 0; k < colsA; k++) {
                    sum += a[i][k] * b[k][j];
                }
                result[i][j] = sum;
            }
        }
        return result;
    }

    private static double[] multiplyVector(double[][] m, double[] v) {
        double[] result = new double[m.length];
        for (int i = 0; i < m.length; i++) {
            double sum = 0.0;
            for (int j = 0; j < v.length; j++) {
                sum += m[i][j] * v[j];
            }
            result[i] = sum;
        }
        return result;
    }

    private static double[][] removeConstantColumns(double[][] X) {

        int d = X[0].length;
        boolean[] keep = new boolean[d];
        int keepCount = 0;

        for (int j = 0; j < d; j++) {
            for (int i = 1; i < X.length; i++) {
                if (Math.abs(X[i][j] - X[0][j]) > EPS) {
                    keep[j] = true;
                    keepCount++;
                    break;
                }
            }
        }

        if (keepCount == d) {
            return X;
        }

        double[][] result = new double[X.length][keepCount];
        for (int i = 0; i < X.length; i++) {
            int newIndex = 0;
            for (int j = 0; j < d; j++) {
                if (keep[j]) {
                    result[i][newIndex] = X[i][j];
                    newIndex++;
                }
            }
        }
        return result;
    }

    private static String[] parseCSV(String line) {

        List<String> fields = new ArrayList<String>();
        StringBuilder current = new StringBuilder();
        boolean inQuotes = false;

        for (int i = 0; i < line.length(); i++) {
            char ch = line.charAt(i);
            if (ch == '"') {
                if (inQuotes && i + 1 < line.length() && line.charAt(i + 1) == '"') {
                    current.append('"');
                    i++;
                } else {
                    inQuotes = !inQuotes;
                }
            } else if (ch == ',' && !inQuotes) {
                fields.add(current.toString());
                current.setLength(0);
            } else {
                current.append(ch);
            }
        }

        fields.add(current.toString());
        return fields.toArray(new String[0]);
    }

    private static String clean(String value) {
        if (value == null) {
            return "";
        }
        return value.replace("\uFEFF", "").replace("'", "").replace("\"", "").trim();
    }

    private static boolean missing(String value) {
        if (value == null || value.trim().isEmpty()) {
            return true;
        }
        String s = value.trim().toLowerCase(Locale.ROOT);
        return s.equals("na") || s.equals("n/a") || s.equals("null")
                || s.equals("nan") || s.equals("?");
    }

    private static Double parseNumber(String value) {
        try {
            String s = value.trim();
            if (s.endsWith("+")) {
                s = s.substring(0, s.length() - 1);
            }
            double d = Double.parseDouble(s);
            if (Double.isFinite(d)) {
                return d;
            }
            return null;
        } catch (NumberFormatException e) {
            return null;
        }
    }

    private static boolean isId(String name) {
        return name.equals("id") || name.equals("index") || name.equals("loan_id")
                || name.equals("loanid") || name.equals("customer_id");
    }

    private static double median(List<Double> values) {
        if (values.isEmpty()) {
            return 0.0;
        }
        Collections.sort(values);
        int n = values.size();
        if (n % 2 == 1) {
            return values.get(n / 2);
        }
        return 0.5 * (values.get(n / 2 - 1) + values.get(n / 2));
    }

    private static boolean containsIgnoreCase(List<String> list, String target) {
        return findIgnoreCase(list, target) != null;
    }

    private static String findIgnoreCase(List<String> list, String target) {
        for (String s : list) {
            if (s.equalsIgnoreCase(target)) {
                return s;
            }
        }
        return null;
    }

    private static int count(int[] y, int target) {
        int c = 0;
        for (int v : y) {
            if (v == target) {
                c++;
            }
        }
        return c;
    }

    private static double sq(double x) {
        return x * x;
    }

    private static double dot(double[] a, double[] b) {
        double sum = 0.0;
        for (int i = 0; i < a.length; i++) {
            sum += a[i] * b[i];
        }
        return sum;
    }

    private static double norm(double[] a) {
        return Math.sqrt(dot(a, a));
    }

    private static double dist(double[] a, double[] b) {
        double sum = 0.0;
        for (int i = 0; i < a.length; i++) {
            sum += sq(a[i] - b[i]);
        }
        return Math.sqrt(sum);
    }

    private static String bits(int value) {
        String s = Integer.toBinaryString(value);
        while (s.length() < QUBITS) {
            s = "0" + s;
        }
        return s;
    }

    private static String f(double value, int digits) {
        return String.format(Locale.US, "%." + digits + "f", value);
    }

    private static String h(String text) {
        if (text == null) {
            return "";
        }
        return text.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;");
    }
